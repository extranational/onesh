# onesh Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Build `onesh`, a uv-based CLI that asks OpenRouter a one-shot question and renders a single markdown answer, retaining conversation context on request.

**Architecture:** Small focused modules behind plain function/class interfaces — `config` (load/merge config + resolve key), `history` (conversations JSON with injected clock/uuid), `llm` (OpenAI-compatible call), `render` (rich output), `cli` (arg parsing + orchestration). The network boundary and the clock/uuid are injected so pure logic is unit-tested deterministically; rendering and the real API call are exercised manually.

**Tech Stack:** Python 3.10+, uv, `openai`, `python-dotenv`, `rich`; `pytest` for tests.

```python
import json
import uuid
from datetime import datetime, timezone
from pathlib import Path


def _default_clock():
    return datetime.now(timezone.utc).isoformat()


def _default_uuid():
    return str(uuid.uuid4())


def _title(conversation):
    for message in conversation["messages"]:
        if message["role"] == "user":
            return message["content"]
    return "(no messages)"


class History:
    def __init__(self, path, *, clock=None, uuid_factory=None):
        self.path = Path(path)
        self._clock = clock or _default_clock
        self._uuid = uuid_factory or _default_uuid
        if self.path.exists():
            self.state = json.loads(self.path.read_text(encoding="utf-8"))
        else:
            self.state = {"conversations": []}

    def new_conversation(self, model):
        now = self._clock()
        conversation = {
            "id": self._uuid(),
            "created_at": now,
            "updated_at": now,
            "model": model,
            "messages": [],
        }
        self.state["conversations"].append(conversation)
        return conversation

    def _ordered(self):
        return sorted(
            self.state["conversations"],
            key=lambda c: c["updated_at"],
            reverse=True,
        )

    def most_recent(self):
        ordered = self._ordered()
        return ordered[0] if ordered else None

    def select(self, index):
        ordered = self._ordered()
        if index < 1 or index > len(ordered):
            raise IndexError(
                f"conversation {index} out of range (valid: 1..{len(ordered)})"
            )
        return ordered[index - 1]

    def append_turn(self, conversation, user_msg, assistant_msg):
        conversation["messages"].append({"role": "user", "content": user_msg})
        conversation["messages"].append({"role": "assistant", "content": assistant_msg})
        conversation["updated_at"] = self._clock()

    def list_recent(self, limit):
        return [
            (i, _title(conversation))
            for i, conversation in enumerate(self._ordered()[:limit], start=1)
        ]

    def save(self):
        self.path.parent.mkdir(parents=True, exist_ok=True)
        self.path.write_text(json.dumps(self.state, indent=2), encoding="utf-8")
```

```python
import dataclasses
import json
import os
from dataclasses import dataclass
from pathlib import Path

from dotenv import load_dotenv

DEFAULT_CONFIG = {
    "base_url": "https://openrouter.ai/api/v1",
    "api_key_env": "OPENROUTER_API_KEY",
    "model": "google/gemini-3.5-flash",
    "system_prompt": (
        "You are a concise terminal assistant. Answer directly and briefly. "
        "When asked for a command, give the command first, then at most one "
        "short line of explanation."
    ),
    "temperature": 0.2,
    "list_limit": 20,
}


@dataclass
class Config:
    base_url: str
    api_key_env: str
    model: str
    system_prompt: str
    temperature: float
    list_limit: int
    http_referer: str | None = None
    x_title: str | None = None


def default_config_dir():
    return Path.home() / ".config" / "onesh"


def load_config(cli_overrides=None, *, config_dir=None, load_env=True):
    config_dir = Path(config_dir) if config_dir else default_config_dir()
    path = config_dir / "config.json"
    if path.exists():
        data = json.loads(path.read_text(encoding="utf-8"))
    else:
        config_dir.mkdir(parents=True, exist_ok=True)
        path.write_text(json.dumps(DEFAULT_CONFIG, indent=2), encoding="utf-8")
        data = dict(DEFAULT_CONFIG)

    merged = {**DEFAULT_CONFIG, **data}
    for key, value in (cli_overrides or {}).items():
        if value is not None:
            merged[key] = value

    if load_env:
        load_dotenv()

    fields = {f.name for f in dataclasses.fields(Config)}
    return Config(**{k: merged[k] for k in fields if k in merged})


def resolve_api_key(config, env=None):
    env = os.environ if env is None else env
    return env.get(config.api_key_env)
```

from openai import OpenAI


class LLMError(Exception):
    """Raised when the LLM API call fails (network, auth, or API error)."""


def build_client(config, api_key):
    return OpenAI(base_url=config.base_url, api_key=api_key)


def complete(client, config, messages):
    full = [{"role": "system", "content": config.system_prompt}, *messages]
    kwargs = {
        "model": config.model,
        "temperature": config.temperature,
        "messages": full,
    }
    headers = {}
    if config.http_referer:
        headers["HTTP-Referer"] = config.http_referer
    if config.x_title:
        headers["X-Title"] = config.x_title
    if headers:
        kwargs["extra_headers"] = headers

    try:
        response = client.chat.completions.create(**kwargs)
    except Exception as exc:  # noqa: BLE001 - normalize any client error
        raise LLMError(str(exc)) from exc

    return response.choices[0].message.content
```

def render_answer(text, console=None):

def render_list(items, console=None):
    console = console or Console()
    if not items:
        console.print("[dim]No conversations yet.[/dim]")
        return
    width = console.width
    for index, title in items:
        prefix = f"{index:>3}  "
        available = max(10, width - len(prefix))
        if len(title) > available:
            title = title[: available - 1] + "…"
        console.print(f"{prefix}{title}")


def render_error(message, console=None):
    console = console or Console(stderr=True)
    console.print(f"[red]error:[/red] {message}")

import json
import sys
from pathlib import Path

from . import llm
from .config import default_config_dir, load_config, resolve_api_key
from .history import History
from .render import render_answer, render_error, render_list


class UsageError(Exception):
    """Raised for malformed command-line input."""


def parse_args(argv):
    continue_ = False
    index = None
    list_ = False
    model = None
    rest = []

    i = 0
    while i < len(argv):
        token = argv[i]
        if token in ("-l", "--list"):
            list_ = True
            i += 1
        elif token in ("-c", "--continue"):
            continue_ = True
            i += 1
            if i < len(argv) and argv[i].isdigit():
                index = int(argv[i])
                i += 1
        elif token == "--model":
            i += 1
            if i >= len(argv):
                raise UsageError("--model requires a value")
            model = argv[i]
            i += 1
        elif token.startswith("--model="):
            model = token.split("=", 1)[1]
            i += 1
        else:
            rest.append(token)
            i += 1

    return {
        "prompt": " ".join(rest) if rest else None,
        "continue_": continue_,
        "index": index,
        "list": list_,
        "model": model,
    }
```

def _history_path(config_dir):
    base = Path(config_dir) if config_dir else default_config_dir()
    return base / "history.json"


def main(argv=None, *, config_dir=None, env=None):
    if argv is None:
        argv = sys.argv[1:]

    try:
        args = parse_args(argv)
    except UsageError as exc:
        render_error(str(exc))
        return 2

    if not args["list"] and not args["prompt"]:
        render_error('no prompt given. usage: onesh "your question"')
        return 2

    try:
        config = load_config({"model": args["model"]}, config_dir=config_dir)
    except (ValueError, OSError) as exc:
        render_error(f"could not read config: {exc}")
        return 1

    history_path = _history_path(config_dir)

    if args["list"]:
        try:
            history = History(history_path)
        except (ValueError, OSError) as exc:
            render_error(f"could not read history ({history_path}): {exc}")
            return 1
        render_list(history.list_recent(config.list_limit))
        return 0

    api_key = resolve_api_key(config, env=env)
    if not api_key:
        render_error(
            f"{config.api_key_env} not set. Add it to .env or your environment, e.g.\n"
            f"  echo {config.api_key_env}=sk-... >> .env"
        )
        return 1

    try:
        history = History(history_path)
    except (ValueError, OSError) as exc:
        render_error(f"could not read history ({history_path}): {exc}")
        return 1

    if not args["continue_"]:
        conversation = history.new_conversation(config.model)
    elif args["index"] is not None:
        try:
            conversation = history.select(args["index"])
        except IndexError as exc:
            render_error(str(exc))
            return 1
    else:
        conversation = history.most_recent()
        if conversation is None:
            render_error("no conversations to continue.")
            return 1

    messages = conversation["messages"] + [{"role": "user", "content": args["prompt"]}]
    client = llm.build_client(config, api_key)
    try:
        text = llm.complete(client, config, messages)
    except llm.LLMError as exc:
        render_error(str(exc))
        return 1

    history.append_turn(conversation, args["prompt"], text)
    history.save()
    render_answer(text)
    return 0

    raise SystemExit(main())
